cmake_minimum_required(VERSION 3.21)
project(DiminaDesktopSDK VERSION 1.9.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Core Network)
if(QT_VERSION_MAJOR EQUAL 5)
  find_package(Qt5 5.15 REQUIRED COMPONENTS Core Network)
else()
  find_package(Qt6 REQUIRED COMPONENTS Core Network)
endif()
add_library(DiminaDesktop STATIC DiminaDesktop.cpp DiminaDesktop.h)
add_library(Dimina::Desktop ALIAS DiminaDesktop)
target_include_directories(DiminaDesktop PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(DiminaDesktop PUBLIC Qt${QT_VERSION_MAJOR}::Core Qt${QT_VERSION_MAJOR}::Network)

option(DIMINA_BUILD_TESTS "Build SDK lifecycle checks" OFF)
if(DIMINA_BUILD_TESTS)
  enable_testing()
  add_executable(dimina-sdk-lifecycle-test tests/lifecycle.cpp)
  target_link_libraries(dimina-sdk-lifecycle-test PRIVATE Dimina::Desktop)
  add_test(NAME dimina-sdk-lifecycle COMMAND dimina-sdk-lifecycle-test)
endif()
