cmake_minimum_required(VERSION 3.22)
project(dimina_bridge_tests C CXX)
get_filename_component(REPO "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${REPO}/cmake/DiminaQuickJS.cmake")
add_executable(bridge_values bridge_values.cpp)
target_compile_features(bridge_values PRIVATE cxx_std_17)
dimina_configure_quickjs(bridge_values ENABLE_DEBUGGER OFF)
enable_testing()
add_test(NAME bridge_values COMMAND bridge_values)

# Optional host N-API integration test for the Harmony conversion code.
set(NODE_INCLUDE_DIR "" CACHE PATH "Directory containing node_api.h")
if(NODE_INCLUDE_DIR)
    add_library(napi_values MODULE napi_values.cpp)
    target_compile_features(napi_values PRIVATE cxx_std_17)
    target_include_directories(napi_values PRIVATE "${NODE_INCLUDE_DIR}" "${CMAKE_CURRENT_LIST_DIR}/compat")
    set_target_properties(napi_values PROPERTIES PREFIX "" SUFFIX ".node")
    if(APPLE)
        target_link_options(napi_values PRIVATE "-undefined" "dynamic_lookup")
    endif()
    dimina_configure_quickjs(napi_values ENABLE_DEBUGGER OFF)
    find_program(NODE_EXECUTABLE node REQUIRED)
    add_test(NAME napi_values COMMAND "${NODE_EXECUTABLE}" --test "${CMAKE_CURRENT_LIST_DIR}/napi_values.test.mjs")
    set_tests_properties(napi_values PROPERTIES ENVIRONMENT "DIMINA_NAPI_MODULE=$<TARGET_FILE:napi_values>")
endif()
