cmake_minimum_required(VERSION 3.22)
project(dimina_wasm_tests C CXX)
get_filename_component(REPO "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${REPO}/cmake/DiminaQuickJS.cmake")
include("${REPO}/cmake/DiminaWasm.cmake")
add_executable(wasm_qjs host.cpp)
dimina_configure_quickjs(wasm_qjs ENABLE_DEBUGGER OFF)
dimina_configure_wasm(wasm_qjs)
target_compile_definitions(wasm_qjs PRIVATE DIMINA_WASM_QUICKJS)
if(APPLE)
    add_executable(wasm_jsc host.cpp)
    dimina_configure_wasm(wasm_jsc)
    target_compile_definitions(wasm_jsc PRIVATE DIMINA_WASM_JSC)
    target_link_libraries(wasm_jsc PRIVATE "-framework JavaScriptCore")
endif()
add_executable(wasm_memory_layout memory_layout.c)
# Internal loader layout checks must use the same feature flags as the engine.
get_directory_property(WAMR_TEST_INCLUDES DIRECTORY "${REPO}/third_party/wamr" INCLUDE_DIRECTORIES)
get_directory_property(WAMR_TEST_DEFINITIONS DIRECTORY "${REPO}/third_party/wamr" COMPILE_DEFINITIONS)
target_include_directories(wasm_memory_layout PRIVATE ${WAMR_TEST_INCLUDES})
target_compile_definitions(wasm_memory_layout PRIVATE ${WAMR_TEST_DEFINITIONS})
target_link_libraries(wasm_memory_layout PRIVATE dimina_wamr)
